  dimension: customer_id {
    type: number
    description: "The natural identification key for the customer."
    sql: ${TABLE}.customer_id ;;
  }

    type: string
    label: "Customer Name"
    sql: ${TABLE}.customer_name ;;
  }
  dimension: email {

    sql: ${TABLE}.email ;;
  }
  dimension: city {
    type: string
    sql: ${TABLE}.city ;;
  }
  dimension: state {
    type: string
    sql: ${TABLE}.state ;;
  }
  dimension: country {
    type: string
    sql: ${TABLE}.country ;;
  }
  dimension: customer_segment {
    type: string
    label: "Customer Segment"
    sql: ${TABLE}.customer_segment ;;
  }

  }

    type: time
    timeframes: [raw, date, week, month, quarter, year]
