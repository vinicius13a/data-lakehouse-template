# =====================================================================
# EXPLORES
# Fact tables are the base; dimensions join via surrogate keys.
# =====================================================================

# ----- Sales Explore -----
# Grain: one row per sales order line
# Use for: revenue, discount, margin, product / store / customer analysis
explore: fact_sales {
  label: "Fact Sales"
  description: "Sales order lines with full dimensional context. Use for revenue, discount and margin analysis."

  join: dim_customer {
    type: left_outer
    relationship: many_to_one
    sql_on: ${fact_sales.fk_customer} = ${dim_customer.sk_customer} ;;
  }
  join: dim_product {
    type: left_outer
    relationship: many_to_one
    sql_on: ${fact_sales.fk_product} = ${dim_product.sk_product} ;;
  }
  join: dim_store {
    type: left_outer
    relationship: many_to_one
    sql_on: ${fact_sales.fk_store} = ${dim_store.sk_store} ;;
  }
  join: dim_calendar {
    type: left_outer
    relationship: many_to_one
    sql_on: ${dim_calendar.sk_date} = ${fact_sales.fk_date_order} ;;
  }
}
